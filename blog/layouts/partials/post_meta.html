{{- /* Overrides the theme's post_meta.html. Date and reading time only (the blog has one author), dates as "2026 9 10", and "&&" between items, as on sophielwang.com/blog. Each item is a .meta-item so the post list can show just the date. */ -}} {{- $items := slice -}} {{- if not .Date.IsZero -}} {{- $items = $items | append (printf `%s` (.Date | time.Format ":date_long") (.Date | time.Format "2006 1 2")) -}} {{- end -}} {{- if (.Param "ShowReadingTime") -}} {{- $items = $items | append (printf `%d min read` .ReadingTime) -}} {{- end -}} {{- delimit $items `&&` | safeHTML -}}
